## Basic Queries

### Get Recent Transactions

Query recent transactions from a specific protocol:

<CodeGroup>
```graphql
query {
  transactions(
    filters: { protocols: ["jupiter_v6"] }
    pagination: { first: 10 }
  ) {
    edges {
      node {
        signature
        slot
        protocolName
        fee
        computeUnits
        blockTime
      }
      cursor
    }
    pageInfo {
      hasNextPage
    }
  }
}
```

```bash curl
curl -X POST "https://api.solixdb.xyz/graphql" \
  -H "Content-Type: application/json" \
  -d '{
    "query": "{ transactions(filters: { protocols: [\"jupiter_v6\"] }, pagination: { first: 10 }) { edges { node { signature slot protocolName fee computeUnits blockTime } cursor } pageInfo { hasNextPage } } }"
  }'
```

```javascript JavaScript
const query = `
  query {
    transactions(
      filters: { protocols: ["jupiter_v6"] }
      pagination: { first: 10 }
    ) {
      edges {
        node {
          signature
          slot
          protocolName
          fee
          computeUnits
          blockTime
        }
        cursor
      }
      pageInfo {
        hasNextPage
      }
    }
  }
`;

const response = await fetch('https://api.solixdb.xyz/graphql', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({ query }),
});

const data = await response.json();
console.log(data);
```
</CodeGroup>

### Fee Analysis by Day of Week

Analyze fees by day of week to identify patterns:

<CodeGroup>
```graphql
query {
  transactions(
    filters: {
      protocols: ["jupiter_v6"]
      dateRange: { start: "2025-01-01", end: "2025-01-31" }
    }
    groupBy: [DAY_OF_WEEK]
    metrics: [COUNT, AVG_FEE, P95_FEE, P99_FEE]
    sort: { field: AVG_FEE, direction: DESC }
  ) {
    edges {
      node {
        dayOfWeek
        count
        avgFee
        p95Fee
        p99Fee
      }
    }
  }
}
```

```javascript JavaScript
const query = `
  query {
    transactions(
      filters: {
        protocols: ["jupiter_v6"]
        dateRange: { start: "2025-01-01", end: "2025-01-31" }
      }
      groupBy: [DAY_OF_WEEK]
      metrics: [COUNT, AVG_FEE, P95_FEE, P99_FEE]
      sort: { field: AVG_FEE, direction: DESC }
    ) {
      edges {
        node {
          dayOfWeek
          count
          avgFee
          p95Fee
          p99Fee
        }
      }
    }
  }
`;

// Map dayOfWeek (0=Sunday, 6=Saturday) to day names
const dayNames = ['Sunday', 'Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday'];

const response = await fetch('https://api.solixdb.xyz/graphql', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({ query }),
});

const data = await response.json();
const results = data.data.transactions.edges.map(edge => ({
  day: dayNames[edge.node.dayOfWeek],
  ...edge.node
}));

console.log(results);
```
</CodeGroup>

### Hourly Protocol Comparison

Compare multiple protocols by hour:

<CodeGroup>
```graphql
query {
  transactions(
    filters: {
      protocols: ["jupiter_v6", "raydium_amm_v3", "pump_fun"]
      dateRange: { start: "2025-01-01", end: "2025-01-31" }
    }
    groupBy: [PROTOCOL, HOUR]
    metrics: [COUNT, AVG_FEE]
    sort: { field: COUNT, direction: DESC }
    pagination: { first: 100 }
  ) {
    edges {
      node {
        protocol
        hour
        count
        avgFee
      }
    }
  }
}
```

```python Python
import requests
import pandas as pd

query = """
  query {
    transactions(
      filters: {
        protocols: ["jupiter_v6", "raydium_amm_v3", "pump_fun"]
        dateRange: { start: "2025-01-01", end: "2025-01-31" }
      }
      groupBy: [PROTOCOL, HOUR]
      metrics: [COUNT, AVG_FEE]
      sort: { field: COUNT, direction: DESC }
      pagination: { first: 100 }
    ) {
      edges {
        node {
          protocol
          hour
          count
          avgFee
        }
      }
    }
  }
"""

response = requests.post(
    'https://api.solixdb.xyz/graphql',
    json={'query': query}
)

data = response.json()
results = [edge['node'] for edge in data['data']['transactions']['edges']]
df = pd.DataFrame(results)

# Pivot for easier analysis
pivot = df.pivot(index='hour', columns='protocol', values='count')
print(pivot)
```

## ML Dataset Export

### Export Training Dataset

Export a dataset for ML model training:

<CodeGroup>
```graphql
mutation {
  exportDataset(
    config: {
      format: PARQUET
      filters: {
        protocols: ["pump_fun"]
        dateRange: { start: "2025-01-01", end: "2025-01-31" }
      }
      columns: [
        "protocol_name"
        "fee"
        "compute_units"
        "success"
        "instruction_type"
        "hour"
        "day_of_week"
        "accounts_count"
      ]
      sampling: { strategy: RANDOM, rate: 0.1 }
      splits: { train: 0.7, test: 0.2, val: 0.1 }
    }
  ) {
    id
    status
    progress
  }
}
```

```javascript JavaScript
// Step 1: Create export job
const exportMutation = `
  mutation {
    exportDataset(
      config: {
        format: PARQUET
        filters: {
          protocols: ["pump_fun"]
          dateRange: { start: "2025-01-01", end: "2025-01-31" }
        }
        columns: [
          "protocol_name"
          "fee"
          "compute_units"
          "success"
          "instruction_type"
          "hour"
          "day_of_week"
          "accounts_count"
        ]
        sampling: { strategy: RANDOM, rate: 0.1 }
        splits: { train: 0.7, test: 0.2, val: 0.1 }
      }
    ) {
      id
      status
      progress
    }
  }
`;

const exportResponse = await fetch('https://api.solixdb.xyz/graphql', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({ query: exportMutation }),
});

const exportData = await exportResponse.json();
const jobId = exportData.data.exportDataset.id;

// Step 2: Poll for completion
async function checkExportStatus(jobId) {
  const statusQuery = `
    query {
      exportJob(id: "${jobId}") {
        id
        status
        progress
        rowCount
        fileSize
        downloadUrl
        error
      }
    }
  `;
  
  const response = await fetch('https://api.solixdb.xyz/graphql', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ query: statusQuery }),
  });
  
  return await response.json();
}

// Poll every 5 seconds
const interval = setInterval(async () => {
  const status = await checkExportStatus(jobId);
  const job = status.data.exportJob;
  
  console.log(`Progress: ${job.progress}%`);
  
  if (job.status === 'COMPLETED') {
    console.log('Export complete! Download URL:', job.downloadUrl);
    clearInterval(interval);
  } else if (job.status === 'FAILED') {
    console.error('Export failed:', job.error);
    clearInterval(interval);
  }
}, 5000);
```
</CodeGroup>

## Query Complexity Management

### Check Complexity Before Execution

Always check query complexity before executing expensive queries:

<CodeGroup>
```graphql
query {
  queryComplexity(
    filters: {
      dateRange: { start: "2025-01-01", end: "2025-01-31" }
      protocols: ["pump_fun"]
    }
    groupBy: [PROTOCOL, HOUR, INSTRUCTION_TYPE]
    metrics: [COUNT, AVG_FEE, P95_FEE, P99_FEE]
  ) {
    score
    estimatedRows
    baseCost
    recommendations
  }
}
```

```javascript JavaScript
async function checkAndExecute(query, variables) {
  // First, check complexity
  const complexityQuery = `
    query {
      queryComplexity(
        filters: $filters
        groupBy: $groupBy
        metrics: $metrics
      ) {
        score
        estimatedRows
        recommendations
      }
    }
  `;
  
  const complexityResponse = await fetch('https://api.solixdb.xyz/graphql', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      query: complexityQuery,
      variables: {
        filters: variables.filters,
        groupBy: variables.groupBy,
        metrics: variables.metrics
      }
    }),
  });
  
  const complexityData = await complexityResponse.json();
  const complexity = complexityData.data.queryComplexity;
  
  if (complexity.score >= 1000) {
    console.error('Query will be rejected:', complexity.recommendations);
    return null;
  }
  
  if (complexity.score >= 500) {
    console.warn('High complexity query:', complexity.recommendations);
  }
  
  // Execute the actual query
  const response = await fetch('https://api.solixdb.xyz/graphql', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ query, variables }),
  });
  
  return await response.json();
}
```
</CodeGroup>

## Pagination Examples

### Fetch All Results

Fetch all results using cursor-based pagination:

<CodeGroup>
```javascript JavaScript
async function fetchAllTransactions(filters) {
  let allTransactions = [];
  let hasNextPage = true;
  let cursor = null;
  
  while (hasNextPage) {
    const query = `
      query GetTransactions($first: Int, $after: String) {
        transactions(
          filters: $filters
          pagination: { first: $first, after: $after }
        ) {
          edges {
            node {
              signature
              slot
              protocolName
              fee
              computeUnits
            }
            cursor
          }
          pageInfo {
            hasNextPage
            endCursor
          }
        }
      }
    `;
    
    const response = await fetch('https://api.solixdb.xyz/graphql', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        query,
        variables: {
          filters,
          first: 1000,
          after: cursor
        }
      }),
    });
    
    const data = await response.json();
    const connection = data.data.transactions;
    
    allTransactions.push(...connection.edges.map(e => e.node));
    hasNextPage = connection.pageInfo.hasNextPage;
    cursor = connection.pageInfo.endCursor;
    
    console.log(`Fetched ${allTransactions.length} transactions...`);
  }
  
  return allTransactions;
}

// Usage
const filters = {
  protocols: ["pump_fun"],
  dateRange: { start: "2025-01-01", end: "2025-01-31" }
};

const allTransactions = await fetchAllTransactions(filters);
console.log(`Total: ${allTransactions.length} transactions`);
```
</CodeGroup>

async function getProtocolMetrics(protocols, dateRange) {
  const query = `
    query {
      ${protocols.map((p, i) => `
        protocol${i}: protocolStats(
          filters: {
            protocols: ["${p}"]
            dateRange: { start: "${dateRange.start}", end: "${dateRange.end}" }
          }
        ) {
          protocolName
          totalTransactions
          totalFees
          averageFee
          averageComputeUnits
        }
      `).join('\n')}
    }
  `;
  
  const response = await fetch('https://api.solixdb.xyz/graphql', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ query }),
  });
  
  const data = await response.json();
  return Object.values(data.data);
}

// Get metrics for multiple protocols
const protocols = ['jupiter_v6', 'raydium_amm_v3', 'pump_fun'];
const metrics = await getProtocolMetrics(protocols, {
