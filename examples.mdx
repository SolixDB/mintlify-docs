## REST API Examples

### Get Recent Jupiter V6 Transactions

Query recent transactions from Jupiter V6:

<CodeGroup>
```bash
curl -X GET "https://api.solixdb.xyz/api/v1/transactions?protocol_name=jupiter_v6&limit=10"
```

```javascript JavaScript
const response = await fetch(
  'https://api.solixdb.xyz/api/v1/transactions?protocol_name=jupiter_v6&limit=10'
);
const data = await response.json();
console.log(data);
```

```python Python
import requests

response = requests.get(
    'https://api.solixdb.xyz/api/v1/transactions',
    params={'protocol_name': 'jupiter_v6', 'limit': 10}
)
print(response.json())
```
</CodeGroup>

### Compare Multiple Protocols

Compare performance across different DEX protocols:

<CodeGroup>
```bash
# Get Jupiter V6 data
curl -X GET "https://api.solixdb.xyz/api/v1/analytics/protocols?protocol_name=jupiter_v6&date_from=2025-07-20"

# Get Raydium data
curl -X GET "https://api.solixdb.xyz/api/v1/analytics/protocols?protocol_name=raydium_amm_v3&date_from=2025-07-20"
```

```javascript JavaScript
async function compareProtocols() {
  const protocols = ['jupiter_v6', 'raydium_amm_v3', 'pump_fun'];
  const results = {};
  
  for (const protocol of protocols) {
    const response = await fetch(
      `https://api.solixdb.xyz/api/v1/analytics/protocols?protocol_name=${protocol}&date_from=2025-07-20`
    );
    results[protocol] = await response.json();
  }
  
  console.log(results);
}

compareProtocols();
```

```python Python
import requests

protocols = ['jupiter_v6', 'raydium_amm_v3', 'pump_fun']
results = {}

for protocol in protocols:
    response = requests.get(
        'https://api.solixdb.xyz/api/v1/analytics/protocols',
        params={
            'protocol_name': protocol,
            'date_from': '2025-07-20'
        }
    )
    results[protocol] = response.json()

print(results)
```
</CodeGroup>

## GraphQL Examples

### Flexible Field Selection

Query only the fields you need:

<CodeGroup>
```bash
curl -X POST "https://api.solixdb.xyz/graphql" \
  -H "Content-Type: application/json" \
  -d '{
    "query": "{ transactions(protocolName: \"jupiter_v6\", limit: 10) { signature protocolName fee blockTime } }"
  }'
```

```javascript JavaScript
const query = `
  query {
    transactions(protocolName: "jupiter_v6", limit: 10) {
      signature
      protocolName
      fee
      blockTime
    }
  }
`;

const response = await fetch('https://api.solixdb.xyz/graphql', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({ query }),
});

const data = await response.json();
console.log(data);
```

```python Python
import requests

query = """
  query {
    transactions(protocolName: "jupiter_v6", limit: 10) {
      signature
      protocolName
      fee
      blockTime
    }
  }
"""

response = requests.post(
    'https://api.solixdb.xyz/graphql',
    json={'query': query}
)
print(response.json())
```
</CodeGroup>

### Export Data for ML Training

Export large datasets for model training:

<CodeGroup>
```bash
curl -X POST "https://api.solixdb.xyz/api/v1/query" \
  -H "Content-Type: application/json" \
  -d '{
    "query": "SELECT * FROM transactions WHERE protocol_name = '\''pump_fun'\'' AND date >= '\''2025-07-01'\'' AND date <= '\''2025-07-31'\'' LIMIT 10000"
  }' > training_data.json
```

```javascript JavaScript
// For large datasets, consider pagination
async function exportTrainingData() {
  const limit = 1000;
  let offset = 0;
  const allData = [];
  
  while (true) {
    const response = await fetch('https://api.solixdb.xyz/api/v1/query', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        query: `SELECT * FROM transactions WHERE protocol_name = 'pump_fun' AND date >= '2025-07-01' AND date <= '2025-07-31' LIMIT ${limit} OFFSET ${offset}`
      }),
    });
    
    const data = await response.json();
    if (!data.data || data.data.length === 0) break;
    
    allData.push(...data.data);
    offset += limit;
  }
  
  // Save to file or process
  console.log(`Exported ${allData.length} records`);
  return allData;
}

exportTrainingData();
```

```python Python
import requests
import json

def export_training_data():
    limit = 1000
    offset = 0
    all_data = []
    
    while True:
        response = requests.post(
            'https://api.solixdb.xyz/api/v1/query',
            json={
                'query': f"SELECT * FROM transactions WHERE protocol_name = 'pump_fun' AND date >= '2025-07-01' AND date <= '2025-07-31' LIMIT {limit} OFFSET {offset}"
            }
        )
        
        data = response.json()
        if not data.get('data') or len(data['data']) == 0:
            break
        
        all_data.extend(data['data'])
        offset += limit
    
    # Save to file
    with open('training_data.json', 'w') as f:
        json.dump(all_data, f)
    
    print(f"Exported {len(all_data)} records")
    return all_data

export_training_data()
```

### Creating an Analytics Dashboard

<Accordion title="Monitor Protocol Performance">
Build real-time dashboards showing protocol metrics:

```javascript
async function getProtocolMetrics(protocolName) {
  const response = await fetch(
    `https://api.solixdb.xyz/api/v1/analytics/protocols?protocol_name=${protocolName}&date_from=2025-07-20`
  );
  return await response.json();
}

// Get metrics for multiple protocols
const protocols = ['jupiter_v6', 'raydium_amm_v3', 'pump_fun'];
const metrics = await Promise.all(
  protocols.map(p => getProtocolMetrics(p))
);

// Display in your dashboard
console.log(metrics);
```
